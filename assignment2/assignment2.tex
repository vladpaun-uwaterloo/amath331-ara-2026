\begin{problem}
Let $(\mathbf{x}_k)_{k\in\mathbb{N}}$ be a sequence in $\mathbb{R}^n$.
Write $\mathbf{x}_k=(x_{k1},\ldots,x_{kn})$, so that $(x_{kj})_{k\in\mathbb{N}}$
is a sequence in $\mathbb{R}$ for each $j=1,\ldots,n$.

\begin{parts}
    \Needspace{8\baselineskip}\item Let $\mathbf{a}=(a_1,\ldots,a_n)\in\mathbb{R}^n$.
    Prove that $\lim_{k\to\infty}\mathbf{x}_k=\mathbf{a}$ if and only if
    $\lim_{k\to\infty}x_{kj}=a_j$ for each $j=1,\ldots,n$.
    (\emph{Don't forget to prove both directions!})

    \begin{answer}
        \begin{proof}
            Suppose $\lim_{k\to\infty}x_k = \vec{a}$. So $\forall\varepsilon > 0:\ \exists N\in\mathbb{N}:\ \forall k\geq N:\ \norm{x_k - \vec{a}} < \varepsilon$.

            Let $\varepsilon > 0$.
            \begin{align*}
                & (x_{kj} - a_j)^2 \geq 0 & \forall j\in[n]\\
                &(x_{kj} - a_j)^2 \leq \sum_{i = 1}^{n}(x_{ki}-a_i)^2 & \forall j\in[n]\\
                &\abs{x_{kj} - a_j} \leq \sqrt{\sum_{i = 1}^{n}(x_{ki}-a_i)^2} = \norm{x_k-\vec{a}} < \varepsilon & \forall j\in[n],\forall k\geq N\\
                & \abs{x_{kj} - a_j} < \varepsilon & \forall j\in[n],\forall k\geq N\\
                & \therefore \lim_{k\to\infty}x_{kj} = a_j & \forall j\in[n]
            \end{align*}

            Suppose $\forall j\in[n]:\ \lim_{k\to\infty}x_{kj} = a_j$. So $\forall j\in[n]:\ \forall\varepsilon > 0:\ \exists N_j\in\mathbb{N}:\ \forall k\geq N_j:\ \abs{x_{kj} - a_j} < \varepsilon$.
            
            Let $\varepsilon' = \frac{\varepsilon}{\sqrt{n}} > 0$. Since $n$ is finite, take $N = \max\{N_j\}$. Then, it holds that:

            \begin{align*}
                 & \abs{x_{kj} - a_j} < \frac{\varepsilon}{\sqrt{n}} & \forall k\geq N,\forall j\in[n]\\
                 & \abs{x_{kj} - a_j}^2 < \frac{\varepsilon^2}{n} & \forall k\geq N,\forall j\in[n]\\
                 & \sum_{j = 1}^{n}(x_{kj} - a_j)^2 < \varepsilon^2 & \forall k\geq N\\
                 & \norm{x_k - \vec{a}} = \sqrt{\sum_{j = 1}^{n}(x_{kj} - a_j)^2} < \varepsilon & \forall k\geq N\\
                 & \therefore \lim_{k\to\infty}x_k = \vec{a}
            \end{align*}
        \end{proof}
    \end{answer}

\begin{problem}
Let $A\subseteq\mathbb{R}^n$. We define the \emph{closure} of $A$, denoted by
$\overline{A}$, to be the subset
\[
    \overline{A}=\{\mathbf{x}\in\mathbb{R}^n:
    B_\varepsilon(\mathbf{x})\cap A\neq\varnothing
    \text{ for all }\varepsilon>0\}.
\]
That is, a point $\mathbf{x}\in\mathbb{R}^n$ lies in the closure of $A$ if
and only if every open ball of any positive radius intersects $A$.

\begin{parts}
    \Needspace{6\baselineskip}\item Show that $A\subseteq\overline{A}$.

    \begin{answer}
        \begin{proof}
            Let $x\in A\subseteq\mathbb{R}^n$.
            \[
                B_\varepsilon(x) = \{y\in \mathbb{R}^n:\ \norm{x - y} < \varepsilon\} \qquad \forall \varepsilon > 0
            \]
            By this deifiniton, we get that $x\in B_\varepsilon(x)$ for all $\varepsilon > 0$.

            Therefore, $x\in B_\varepsilon(x) \cap A \neq \varnothing$. So $x\in \overline{A}$.

            Therefore, $A\subseteq \overline{A}$.
        \end{proof}
    \end{answer}

    \Needspace{7\baselineskip}\item Show that
    $\overline{A}=\{\mathbf{x}\in\mathbb{R}^n:U\cap A\neq\varnothing
    \text{ for any open set }U\text{ containing }\mathbf{x}\}$.

    \begin{answer}
        \begin{proof} Let arbitrary $x\in\mathbb{R}^n$.
            We want to prove that $\forall \varepsilon > 0:\ B_\varepsilon(x)\cap A\neq\varnothing \iff U\cap A \neq \varnothing$ for all open sets $U$ containing $x$.

            First, prove the forward direction:

            Suppose $\forall \varepsilon > 0:\ B_\varepsilon(x)\cap A\neq\varnothing$.\\
            Take an arbitrary open set $U\subseteq\mathbb{R}^n$ containing $x$. Since $U$ is open:
            \[
                \exists \delta > 0:\ B_\delta(x)\subseteq U
            \]
            By our assumption, we know that $B_\delta(x)\cap A\neq\varnothing$. So take arbitrary $y\in B_\delta(x)\cap A$. \\
            So $y\in A$ and $y\in B_\delta(x)$. But since $B_\delta(x)\subseteq U$, we get that $y\in U$.\\
            Therefore, $y\in U\cap A$, so $U\cap A \neq\varnothing$. Since $U$ is arbitrary, it holds for all open sets containing $x$.

            Now, for the reverse direction:

            Suppose $U\cap A \neq \varnothing$ for all open sets $U$ containing $x$.\\
            Let arbitrary $\varepsilon > 0$ and take the open set $U = B_\varepsilon(x)$ containing $x$. Then, we have:
            \[\forall \varepsilon > 0: B_\varepsilon(x) \cap A \neq \varnothing\]

            Therefore, we have equivalence of the 2 statements. So:

            \begin{align*}
                &\overline{A}=\{\mathbf{x}\in\mathbb{R}^n:
                B_\varepsilon(\mathbf{x})\cap A\neq\varnothing
                \text{ for all }\varepsilon>0\} \iff \\
                & \overline{A}=\{\mathbf{x}\in\mathbb{R}^n:U\cap A\neq\varnothing
                \text{ for any open set }U\text{ containing }\mathbf{x}\}
            \end{align*}
        \end{proof}
    \end{answer}

    \Needspace{9\baselineskip}\item Show that $\overline{A}$ is closed.

    \emph{Hint:} First show there exists $\varepsilon_0$ such that
    $B_{\varepsilon_0}(\mathbf{x})\cap A=\varnothing$, and then show that
    $B_{\varepsilon_0}(\mathbf{x})\cap\overline{A}=\varnothing$.
    You will need to use part [b].

    \begin{answer}
        \begin{proof}
            Let's look at $\overline{A}^c = \mathbb{R}^n \setminus \overline{A}$.
            \[\forall x\in\overline{A}^c:\ \exists \varepsilon > 0:\ B_\varepsilon(x) \cap A = \varnothing\]
            Take arbitrary $x\in\overline{A}^c$. Then $\varepsilon_0$ exists such that $B_{\varepsilon_0}(x) \cap A = \varnothing$.

            Now assume for the sake of contradiction that there exists $y$ such that 
            \[y\in B_{\varepsilon_0}(x)\cap\overline{A}\]
            $B_{\varepsilon_0}(x)$ is an open set containing $y$ and $y\in\overline{A}$, so $B_{\varepsilon_0}(x)\cap A \neq \varnothing$, which is a contradiction.

            Therefore, $B_{\varepsilon_0}(x)\cap\overline{A} = \varnothing$. Since $x$ is arbitrary, it holds that
            \begin{align*}
                &\forall x\in\overline{A}^c:\ \exists \varepsilon > 0:\ B_{\varepsilon}(x)\cap\overline{A} = \varnothing\\
                &\forall x \in\overline{A}^c:\ \exists \varepsilon > 0:\ B_\varepsilon(x) \subseteq (\mathbb{R}^n\setminus\overline{A})\\
                &\forall x \in\overline{A}^c:\ \exists \varepsilon > 0:\ B_\varepsilon(x) \subseteq \overline{A}^c\\
                & \therefore \overline{A}^c \text{ is open}
            \end{align*}
            Therefore, $\overline{A}$ is closed.
        \end{proof}
    \end{answer}

    \Needspace{6\baselineskip}\item If $A$ is closed, show that $\overline{A}=A$.

    \begin{answer}
        \begin{proof}
            Assume $A$ is closed. We know that $A\subseteq\overline{A}$, so we have to show $\overline{A}\subseteq A$.

            Take arbitrary $y\in\overline{A}$.
            \[\forall \varepsilon > 0:\ B_\varepsilon(y) \cap A \neq \varnothing\]

            We know $A^c$ is open:
            \[\forall x\in A^c:\ \exists \varepsilon > 0: B_\varepsilon(x) \subseteq A^c\]
            Which is equivalent to:
            \[\forall x\in A^c:\ \exists \varepsilon > 0: B_\varepsilon(x) \cap A = \varnothing\]

            Now, assume for the sake of contradiction that $y\in A^c$. 
            Since \(A^c\) is open and \(y\in A^c\), there exists \(\varepsilon_0>0\) such that
            \[
            B_{\varepsilon_0}(y)\subseteq A^c
            \]
            \[
            B_{\varepsilon_0}(y)\cap A=\varnothing.
            \]But since \(y\in\overline A\)
            \[
            B_{\varepsilon_0}(y)\cap A\neq\varnothing
            \]a contradiction. Therefore,
            \begin{align*}
                &y\not\in A^c\\
                &y\in A
            \end{align*}
            Since $y$ is arbitrary, it holds for all $y\in\overline{A}$. Therefore, $\overline{A}\subseteq A$.
        \end{proof}
    \end{answer}
\end{parts}
(Combining parts [c] and [d] shows that $A$ is closed if and only if

            \[B_\varepsilon(\vec{p})\cap E \neq \varnothing\]
            Since $\varepsilon$ was arbitrary, it holds that
            \[\forall \varepsilon > 0:\ B_\varepsilon(\vec{p})\cap E \neq \varnothing\]

            Which is exactly the condition for being a member of $\overline{E}$,

            Therefore, $\vec{p}\in \overline{E}$. 
        \end{proof}
    \end{answer}

    \Needspace{8\baselineskip}\item Let $\mathbf{p}\in\overline{E}$.
    Prove that there exists a sequence $(\mathbf{x}_k)_{k\in\mathbb{N}}$ such
    that $\mathbf{x}_k\in E$ for all $k\in\mathbb{N}$, and
    $\lim_{k\to\infty}\mathbf{x}_k=\mathbf{p}$.

    \begin{answer}
        \begin{proof}
            Since $\vec{p}\in\overline{E}$
            \[\forall \varepsilon > 0:\ B_\varepsilon(\vec{p})\cap E \neq \varnothing\]

            Consider the values $\varepsilon = \frac{1}{k}$ for all $k\in \mathbb{N}$. We know that for each of these values, there is at least one element both in $E$ and $B_\frac{1}{k}(\vec{p})$. Call that element $x_k$.

            More formally:
            \[\forall k\in\mathbb{N}:\ x_k\in B_\frac{1}{k}(\vec{p}) \wedge x_k\in E\]
            This is a sequence $(x_k)_{k\in\mathbb{N}}$ with $x_k\in E$ for every $k\in\mathbb{N}$.

\begin{problem}
Let $(\mathbf{x}_m)_{m\in\mathbb{N}}$ be a sequence in $\mathbb{R}^n$.
Define a new sequence $(\mathbf{s}_k)_{k\in\mathbb{N}}$ by
\[
    \mathbf{s}_k=\sum_{m=1}^k\mathbf{x}_m.
\]
If the sequence $(\mathbf{s}_k)$ converges, then we say that the \emph{series}
$\sum_{m=1}^{\infty}\mathbf{x}_m$ \emph{converges}.
If $\lim_{k\to\infty}\mathbf{s}_k=\mathbf{s}$, then we write
$\sum_{m=1}^{\infty}\mathbf{x}_m=\mathbf{s}$.

\begin{parts}
    \Needspace{12\baselineskip}\item Prove that
    $\sum_{m=1}^{\infty}\mathbf{x}_m$ converges iff: for all $\varepsilon>0$,
    there exists $N\in\mathbb{N}$ such that
    \[
        \left\lVert\sum_{m=k+1}^{\ell}\mathbf{x}_m\right\rVert<\varepsilon
    \]
    for all $\ell\geq k\geq N$.

    \emph{Hint:} A sequence in $\mathbb{R}^n$ converges if and only if it is Cauchy.

    \begin{answer}
        \begin{proof}
            \begin{align*}
                &\forall\varepsilon>0:\ \exists N\in\mathbb{N}:\ \forall l\geq k\geq N:\ \norm{\sum_{m=k+1}^{\ell}\mathbf{x}_m}<\varepsilon\\
                \iff &\forall\varepsilon>0:\ \exists N\in\mathbb{N}:\ \forall l\geq k\geq N:\ \norm{\sum_{m=1}^{\ell}\mathbf{x}_m - \sum_{m=1}^{k}\mathbf{x}_m}<\varepsilon\\
                \iff &\forall\varepsilon>0:\ \exists N\in\mathbb{N}:\ \forall l\geq k\geq N:\ \norm{s_\ell - s_k}<\varepsilon\\
                \iff & (s_k)_{k\in\mathbb{N}} \text{ is Cauchy}\\
                \iff & (s_k)_{k\in\mathbb{N}} \text{ converges.}
            \end{align*}
        \end{proof}
    \end{answer}

    \Needspace{13\baselineskip}\item Suppose that
    $\sum_{m=1}^{\infty}\mathbf{x}_m$ converges. For any $N\in\mathbb{N}$,
    prove that $\sum_{m=N}^{\infty}\mathbf{x}_m$ converges, and that
    \[
        \lim_{N\to\infty}\left(\sum_{m=N}^{\infty}\mathbf{x}_m\right)=0.
    \]
    \emph{Hint:} Consider the sequence
    $(\tilde{\mathbf{s}}_\ell)_{\ell\geq N}$ given by
    $\tilde{\mathbf{s}}_\ell=\sum_{m=N}^{\ell}\mathbf{x}_m$.

    \begin{answer}
        \begin{proof}
            Consider the sequence $(\tilde{s}_\ell)_{\ell\geq N}$ given by $\tilde{s}_\ell=\sum_{m=N}^{\ell}x_m$.\

            Since we know $s_\ell$ converges, let $\lim_{\ell\to\infty}s_\ell =  \sum_{m=1}^{\infty}x_m = p$. 
            
            Also take arbitrary $\varepsilon > 0$.

            \[\exists L\in\mathbb{N}:\ \forall \ell \geq L:\ \norm{s_\ell - p} < \varepsilon\]
            Now,
            \[
                \forall \ell \geq L:\ \norm{\tilde{s}_\ell - (p - s_{N-1})} = \norm{s_\ell - p} < \varepsilon
            \]
            Therefore, $(\tilde{s}_\ell)_{l\geq N}$ converges to $p - s_{N-1}$.
            \[\sum_{m=N}^{\infty}x_m = p - s_{N-1}\]
            Let arbitrary $\varepsilon>0$. Since $s_k\to p$, there exists $K\in\mathbb{N}$ such that
            \[
            \forall k\geq K:\ \norm{s_k-p}<\varepsilon.
            \]

            Take $N\geq K+1$. Then $N-1\geq K$, and therefore
            \begin{align*}
                \norm{\sum_{m=N}^{\infty}x_m - 0} &=\norm{p-s_{N-1}} <\varepsilon.
            \end{align*}
    
            Since $\varepsilon>0$ was arbitrary,
            \[
                \lim_{N\to\infty}
                \left(
                    \sum_{m=N}^{\infty}x_m
                \right)
                =0.
            \]
        \end{proof}
    \end{answer}
\end{parts}
\end{problem}

\begin{problem}

\begin{answer}
    \begin{proof} Let $f:\mathbb{N}\to\mathbb{N}$ strictly increasing. And denote $(x_{f(k)})_{k\in\mathbb{N}}$ a subsequence of $(x_k)$.

        Firstly, prove:
        \[\Big(\lim_{k\to\infty}x_{f(k)} = \vec{x}\Big) \implies
        \Big(\forall \varepsilon > 0:\ x_k\in B_\varepsilon(\vec{x})\text{ for infinitely many } k \Big)

        Assume $\lim_{k\to\infty}x_{f(k)} = \vec{x}$.
        Let arbitrary $\varepsilon > 0$. Take $K\in\mathbb{N}$ such that $\forall k \geq K:\ \norm{\vec{x} - x_{f(k)}} < \varepsilon$.
        \[\forall k\geq K:\ x_{f(k)}\in B_\varepsilon(\vec{x})\]
        We know from the archimedean property that there are an infinite amount of $k\geq K$, and since $f$ is strictly incresing, all $f(k)$ are distinct, so there is an infinite amount of $x_{f(k)}$ such that $k\geq K$.

        Since $\varepsilon$ was arbitrary, it holds for all $\varepsilon > 0$.

        Secondly, prove:
        \[\Big(\forall \varepsilon > 0:\ x_k\in B_\varepsilon(\vec{x})\text{ for infinitely many } k \Big)
        \implies \big(\lim_{k\to\infty}x_{f(k)} = \vec{x}\big)\]
        Assume $\forall \varepsilon > 0:\ x_k\in B_\varepsilon(\vec{x})\text{ for infinitely many } k$.
        
        Now, for every $j\in\mathbb{N}$ do the following:
        \begin{itemize}
            \item Apply the assumption with $\varepsilon = \frac{1}{j}$: $x_k\in B_\frac{1}{j}(\vec{x})\text{ for infinitely many } k$.
            \item Choose $f(j) = k_j$, such that:
            \begin{itemize}
                \item $x_{k_j}\in B_\frac{1}{j}(\vec{x})$.
                \item For $j > 1$: $f(j) > f(j-1)$ (this can always be satisfied, as for every $j$ we have an infinite amount of $k_j$ to choose from)
            \end{itemize}
        \end{itemize}
        With this construction of $f(k)$, we have:
        \[\forall k\in\mathbb{N}:\ \norm{x_{f(k)}-\vec{x}} < \frac{1}{k}\] 
        $\lim_{k\to\infty}\frac{1}{k} = 0$, therefore $\lim_{k\to\infty}x_{f(k)} = \vec{x}$
    \end{proof}
\end{answer}
